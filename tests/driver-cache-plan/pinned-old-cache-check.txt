cache_check() {  # cache_check: sets cache_entry (empty when the cache does not apply) and exits 0 on a hit
cache_entry=""
[ -n "$cache_dir" ] || return 0   # the cache is off: no Ruby is started
local f a look decision state key pre=() cpp=() opts=() key_argv=()
for a in ${nfcxx_argv[@]+"${nfcxx_argv[@]}"}; do
  case $a in --cache=*|--cache-log=*|--time-log=*|--trace=*) ;; *) key_argv+=("$a") ;; esac
done
for f in ${preinc[@]+"${preinc[@]}"}; do pre+=(-include "$f"); done
# the host preprocessor's arguments (its -M command line, after -M and before the source)
cpp=("-std=${std#--}" ${cflags[@]+"${cflags[@]}"} ${pre[@]+"${pre[@]}"} ${sysdirs[@]+"${sysdirs[@]/#/-isystem}"})
opts=(--cache-dir="$cache_dir" --cache-log="$cache_log_file" --out="$out" --src="${srcs[0]-}" --cwd="$PWD"
      --var=backend="$backend" --var=compile_only="$compile_only" --var=ncpp="$ncpp" --var=outset="$outset"
      --var=nobj="$nobj" --var=modules="$modules" --var=preprocess="$preprocess" --var=stdin_src="$stdin_src"
      --var=depmode="$depmode" --var=cfiles_n="${#cfiles[@]}" --var=cu_n="${#cu_inputs[@]}" --var=cpp2_n="${#cpp2_inputs[@]}")
for a in "${key_argv[@]+"${key_argv[@]}"}"; do opts+=(--arg="$a"); done
for a in "${cpp[@]+"${cpp[@]}"}"; do opts+=(--cpp="$a"); done
# lookup: the --cache-log block (when there is a log) and the last line, "hit KEY" or "miss KEY"; nothing when not cached
look=$("${RUBY:-ruby}" "$root/scripts/nfcxx-cache.rb" lookup "${opts[@]}") || return 0
[ -n "$look" ] || return 0
decision=${look##*$'\n'}
[ "$look" = "$decision" ] || cache_log "${look%$'\n'*}"
read -r state key <<<"$decision"
cache_entry=$cache_dir/objects/$key
# a local miss is looked up in the remote tier (nfcxx-cache-io.rb remote-get; a failure is a miss)
if [ "$state" = miss ]; then
  [ -z "$cache_remote" ] || cache_remote_get "$key" "$cache_entry" || true
  [ -f "$cache_entry/obj" ] || return 0
fi
# hit: the object and its records are copied back, and the program is done
"${RUBY:-ruby}" "$root/scripts/nfcxx-cache.rb" hit --out="$out" --entry="$cache_entry" && exit 0
return 0
}
